cmake_minimum_required(VERSION 3.20)
project(ntc LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

function(ntc_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /EHsc)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wno-unused-parameter -Wno-missing-field-initializers)
    endif()
endfunction()

add_library(ntc STATIC cpp/src/ntc.cpp third_party/puff/puff.c)
target_include_directories(ntc PUBLIC cpp/include ${CMAKE_SOURCE_DIR}/third_party ${CMAKE_SOURCE_DIR}/third_party/puff)
if(MSVC)
    set_source_files_properties(third_party/puff/puff.c PROPERTIES COMPILE_OPTIONS "/w")
else()
    set_source_files_properties(third_party/puff/puff.c PROPERTIES COMPILE_OPTIONS "-w")
endif()
ntc_warnings(ntc)

add_executable(ntc_decode cpp/src/decode_main.cpp)
target_link_libraries(ntc_decode PRIVATE ntc)
ntc_warnings(ntc_decode)

add_executable(test_ntc cpp/tests/test_ntc.cpp)

add_executable(expcode_bytes cpp/src/expcode_bytes.cpp)

find_program(GLSLANG glslangValidator)
set(SHADER_DIR ${CMAKE_SOURCE_DIR}/shaders)
set(SPV_DIR ${CMAKE_BINARY_DIR}/shaders)
file(MAKE_DIRECTORY ${SPV_DIR})

function(ntc_spirv src env out_name)
    if(NOT GLSLANG)
        return()
    endif()
    add_custom_command(
        OUTPUT ${SPV_DIR}/${out_name}
        COMMAND ${GLSLANG} -V --target-env ${env} "-I${SHADER_DIR}"
                ${SHADER_DIR}/${src} -o ${SPV_DIR}/${out_name}
        DEPENDS ${SHADER_DIR}/${src} ${SHADER_DIR}/ntc_decode.glsl
        VERBATIM)
    set(NTC_SPV_OUTPUTS ${NTC_SPV_OUTPUTS} ${SPV_DIR}/${out_name} PARENT_SCOPE)
endfunction()
